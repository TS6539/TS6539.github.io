{{/* Hugo Blox: Citation Impact (custom block) */}} {{/* Renders a stats row, a Leaflet world map of citing institutions, and a citations-per-year chart from data/citations.json (baked in at build time). The citation story can play back chronologically: institutions appear by first citing year while the counters count up in sync. The animation is progressive enhancement on top of the full static render. */}} {{ $page := .wcPage }} {{ $block := .wcBlock }} {{ $content := $block.content }} {{ if not (reflect.IsMap $content) }}{{ $content = dict }}{{ end }} {{ $title := index $content "title" | default "Impact" }} {{ $data := hugo.Data.citations }} {{ $totals := $data.totals }} {{/* Chronological playback steps, computed from the data at build time. Institutions are grouped by first citing year (year_min); the one institution without a year joins the final group so the end state matches the static totals. Countries are the cumulative distinct country codes up to and including each year. */}} {{ $cyp := $data.citations_per_year }} {{ $lastIdx := sub (len $cyp) 1 }} {{ $steps := slice }} {{ range $i, $cy := $cyp }} {{ $count := 0 }} {{ $codes := slice }} {{ range $data.institutions }} {{ $inc := false }} {{ if .year_min }} {{ if le (int .year_min) (int $cy.year) }}{{ $inc = true }}{{ end }} {{ else if eq $i $lastIdx }} {{ $inc = true }} {{ end }} {{ if $inc }} {{ $count = add $count 1 }} {{ $codes = $codes | append .country_code }} {{ end }} {{ end }} {{ $steps = $steps | append (dict "year" (int $cy.year) "papers" (int $cy.cumulative) "institutions" $count "countries" (len ($codes | uniq))) }} {{ end }}
